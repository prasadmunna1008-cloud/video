import os from moviepy.editor import AudioFileClip, ImageClip print("🎬 कॉमेडी वीडियो टेस्टिंग इंजन शुरू हो रहा है...") # फाइलों के नाम audio_file = "voice.mp3" background_image = "shop.jpg" output_video = "final_output.mp4" # 1. ऑडियो लोड करना audio_clip = AudioFileClip(audio_file) duration = audio_clip.duration print(f"🎵 टेस्ट ऑडियो मिल गया! लंबाई: {duration} सेकंड") # 2. बैकग्राउंड इमेज को 1080x1920 (यूट्यूब शॉर्ट्स साइज) में सेट करना base_clip = ImageClip(background_image).set_duration(duration) base_clip = base_clip.resize(newsize=(1080, 1920)) # 3. ऑडियो और वीडियो को मर्ज करना final_video = base_clip.set_audio(audio_clip) # 4. HIGH QUALITY 1080p रेंडरिंग print("⏳ वीडियो जनरेट हो रही है... बस 1-2 मिनट रुकें...") final_video.write_videofile( output_video, fps=30, codec="libx264", audio_codec="aac", bitrate="5000k" ) print("🎉 बधाई हो! टेस्ट वीडियो पूरी तरह तैयार है!")
